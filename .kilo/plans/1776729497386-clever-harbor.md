# Plan: Auto-Generating Timeline

## Goal
Automatically generate the timeline table from content collections instead of manually maintaining `timeline.mdx`.

## High-Level Approach

1. **Add `timeline` frontmatter field** — Allow content files to opt-in/out of timeline
2. **Create timeline library** — Extract dates from content, sort, generate MDX
3. **Hook into Astro build** — Generate timeline file at build time
4. **Refactor timeline page** — Import generated table

## Detailed Technical Specification

### 2. Timeline Library (`src/lib/timeline.ts`)

Create new file with the following exports:

```typescript
/**
 * TimelineEntry represents a single row in the timeline.
 */
interface TimelineEntry {
  year: string;           // Raw string: "460 PF", "1255 AI"
  yearSort: number;       // Normalized: negative=PF, positive=AI
  label: string;         // Display text (title OR timeline.label)
  slug: string;           // entry.slug from getCollection()
  collection: 'events' | 'characters' | 'organizations' | 'locations';
  type: 'birth' | 'death' | 'start' | 'end';
  priority: number;      // For same-year tiebreaking (lower = first)
}

// href is computed in generateTimelineMDX() from collection + slug:
// '/world/events/some-event/' from collection 'events' + slug 'some-event'

/**
 * normalizeYear converts date string to sortable number.
 *   "3000 PF"  → -3000
 *   "1 AI"     → 1
 *   "1255-07-25 AI" → 1255 (uses year part only)
 */
function normalizeYear(dateStr: string): number {
  const match = dateStr.match(/^(\d+)(?:-(\d+)-(\d+))?\s*(PF|AI)$/i);
  if (!match) return 0;
  const [, year, , , era] = match;
  const num = parseInt(year, 10);
  return era.toUpperCase() === 'PF' ? -num : num;
}

/**
 * getDateField extracts date string from entry based on collection type.
 * Returns null if no date field exists or timeline is false.
 */
function getDateField(
  collection: string,
  entry: { data: Frontmatter }
): { date: string; type: 'birth' | 'death' | 'start' | 'end' } | null {
  const data = entry.data as any;
  const timeline = data.timeline;

  // Explicit timeline setting takes precedence
  if (timeline === false) return null;
  if (typeof timeline === 'object' && timeline?.year) {
    return { date: timeline.year, type: timeline.type || 'start' };
  }

  // Default based on collection
  switch (collection) {
    case 'events':
      return data.event?.dateStarted 
        ? { date: data.event.dateStarted, type: 'start' }
        : data.event?.dateEnded
        ? { date: data.event.dateEnded, type: 'end' }
        : null;
    case 'characters':
      return data.character?.details?.dateOfBirth
        ? { date: data.character.details.dateOfBirth, type: 'birth' }
        : data.character?.details?.dateOfDeath
        ? { date: data.character.details.dateOfDeath, type: 'death' }
        : null;
    case 'organizations':
      return data.organization?.founded
        ? { date: data.organization.founded, type: 'start' }
        : data.organization?.dissolved
        ? { date: data.organization.dissolved, type: 'end' }
        : null;
    case 'locations':
      return data.location?.details?.timeline?.start
        ? { date: data.location.details.timeline.start, type: 'start' }
        : data.location?.details?.timeline?.end
        ? { date: data.location.details.timeline.end, type: 'end' }
        : null;
    default:
      return null;
  }
}

/**
 * getTimelineEntries queries all collections and returns sorted entries.
 */
async function getTimelineEntries(): Promise<TimelineEntry[]> {
  const { getCollection } = await import('astro:content');
  
  const entries: TimelineEntry[] = [];
  
  // Query each collection
  const collections = ['events', 'characters', 'organizations', 'locations'] as const;
  
  for (const coll of collections) {
    const collEntries = await getCollection(coll);
    
    for (const entry of collEntries) {
      const timeline = (entry.data as any).timeline;
      
      // Skip if explicitly false
      if (timeline === false) continue;
      
      // Default: events include, others exclude (unless timeline: true)
      if (timeline === undefined && coll !== 'events') continue;
      if (timeline === true && coll !== 'events') {
        // Use default date fields for opt-in collections
      }
      
      const dateInfo = getDateField(coll, entry);
      if (!dateInfo && timeline !== true) continue;
      
      // Get custom values if provided
      const custom = typeof timeline === 'object' ? timeline : null;
      
      const year = custom?.year || dateInfo?.date || '';
      if (!year) continue;
      
      const label = custom?.label || entry.data.title || entry.id;
      const slug = entry.slug || entry.id;  // Use entry.slug from collection
      const priority = custom?.priority ?? custom?.order ?? 999;
      
      entries.push({
        year,
        yearSort: normalizeYear(year),
        label,
        slug,
        collection: coll,
        type: custom?.type || dateInfo?.type || 'start',
        priority,
      });
    }
  }
  
  // Sort: by yearSort, then by priority
  entries.sort((a, b) => {
    if (a.yearSort !== b.yearSort) return a.yearSort - b.yearSort;
    return a.priority - b.priority;
  });
  
  return entries;
}

/**
 * generateTimelineMDX creates the MDX table content.
 */
function generateTimelineMDX(entries: TimelineEntry[]): string {
  // Map collection to path segment
  const pathMap = {
    events: 'world/events',
    characters: 'world/characters',
    organizations: 'world/organizations',
    locations: 'world/locations',
  };

  const rows = entries.map(e => {
    const href = `/${pathMap[e.collection]}/${e.slug}/`;
    return `    <tr>\n      <td>${e.year}</td>\n      <td>[${e.label}](${href})</td>\n    </tr>`;
  }).join('\n');

  return `---
title: Timeline Generated
---

<table>
  <thead>
    <tr>
      <th>Year</th>
      <th>Event</th>
    </tr>
  </thead>
  <tbody>
${rows}
  </tbody>
</table>
`;
}

export { TimelineEntry, getTimelineEntries, generateTimelineMDX };
```

### 3. Build Hook (`astro.config.mjs`)

Add hook to generate timeline at build start:

```javascript
import { writeFileSync } from 'node:fs';
// ... existing imports

// Add at end of config object:
hooks: {
  'astro:build:start': async () => {
    const { getTimelineEntries, generateTimelineMDX } = await import('./src/lib/timeline.ts');
    const entries = await getTimelineEntries();
    const mdx = generateTimelineMDX(entries);
    writeFileSync(
      './src/content/docs/world/timeline/timeline-generated.mdx',
      mdx
    );
  },
},
```

## Usage Examples

### Example 1: Event (default included)

```yaml
---
title: Fall of Bastion
tags:
  - events
event:
  dateStarted: '1225-06-12 AI'
---
# Automatically in timeline
```

```yaml
---
title: Narmaya
tags:
  - characters
timeline: true

  details:
    dateOfBirth: '876 AI'

---
# Narmaya appears in timeline (both birth and death)
```

## Important Implementation Notes

1. The `astro:content` import must be dynamic (`await import`) because content collections are only available at build time, not at config evaluation time.

2. The `docsExtension` is shared across all content, so `timeline` field works on any content type.

3. Collection paths for href generation:
   - Events: `/world/events/{slug}/`
   - Characters: `/world/characters/{slug}/`
   - Organizations: `/world/organizations/{slug}/`
   - Locations: `/world/locations/{slug}/`

4. Date format parsing handles: `YYYY PF`, `YYYY AI`, `YYYY-MM-DD AI`, etc.

5. Priority defaults to 999 for tiebreaking; explicit priorities 1-100 for important events can be set manually.
